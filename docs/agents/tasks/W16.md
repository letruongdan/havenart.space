## Code và test cho model nhẹ

VisibilityInput do W02 khai báo trong types/runtime; helper không nhân bản type.

```ts
import type { VisibilityInput } from "../../types/runtime";
export function isHotspotVisible(input: VisibilityInput): boolean {
  return Number.isFinite(input.distanceM) && Number.isFinite(input.maxDistanceM)
    && input.distanceM >= 0 && input.maxDistanceM > 0
    && input.activeRoom === input.room && input.inActivationRange
    && input.distanceM <= input.maxDistanceM
    && input.inFrustum && !input.occluded;
}
```

```ts
import { expect, test } from "vitest";
import { isHotspotVisible } from "../../src/lib/story/hotspotVisibility";
const valid = {activeRoom:"living",room:"living",distanceM:2,maxDistanceM:3,inFrustum:true,occluded:false,inActivationRange:true};
test("all spatial conditions hold", () => {
  expect(isHotspotVisible(valid)).toBe(true);
  expect(isHotspotVisible({...valid,distanceM:3})).toBe(true);
});
test.each([
  {activeRoom:"garden"},{inActivationRange:false},{distanceM:4},
  {distanceM:NaN},{distanceM:-1},{maxDistanceM:Infinity},{maxDistanceM:0},
  {inFrustum:false},{occluded:true}
])("rejects a broken condition %j", patch => expect(isHotspotVisible({...valid,...patch})).toBe(false));
```
